Add a Dynamic Form and use the Configure as Edit Form quick start to display employee details.

Customize a Dropdown List to use a Table Item Layout

  1. Create an item layout departmentItemTable in the top-level layouts section of the layout.json file with "displayType": "table". 
      "departmentItemTable": {
    "description": "layout for department table",
    "layoutType": "item",
      "layout": {
        "displayValue": [
          "departmentName"
        ],
        "displayType": "table",
        "displayProperties": [
          "departmentName",
          "totalSalary"
        ]
      }
  },

  2. In the same layout.json file, under the display properties of the layout for the dynamic form, set the itemLayout property of the dropdown list field to the item layout departmentItemTable created in the previous step.

    In the following code snippet, itemLayout is set for department field. 

        "displayProperties": [
      "firstName",
      "lastName",
      {
        "department": {
          "itemLayout": "departmentItemTable"
        }
      },

Customize a Dropdown List to use a List Item Layout

  1. Create an item layout departmentItemList in the top-level layouts section of layout.json file 
        "departmentItemList": {
      "description": "layout for department item",
      "layoutType": "item",
      "layout": {
        "templateId" : "listItemLayoutTemplate"
      }
    },

  2. In the same layout.json file, under the display properties of the layout for the dynamic form, set the itemLayout property of the dropdown list field to the item layout departmentItemList created in the previous step.

    In the below code snippet, itemLayout is set for department field. 

        "displayProperties": [
      "firstName",
      {
        "department": {
          "itemLayout": "departmentItemList"
        }
      },
  3. Create a field template for the dropdown field, and configure the oj-list-item-layout component. 
    <template id="listItemLayoutTemplate">
  <oj-list-item-layout class="oj-listitemlayout-padding-off">
    <span class="oj-typography-body-lg oj-text-color-primary">
      <oj-bind-text value="[[$fields.departmentName.$value()]]"></oj-bind-text>
    </span>
    <span slot="secondary" class="oj-typography-body-md oj-text-color-secondary">
      <oj-bind-text value="[[ $fields.locationObject.$value()?.items[0]?.city ]]">
      </oj-bind-text>
    </span>
    <div slot="metadata" class="oj-typography-body-sm oj-text-color-secondary">
      <span><oj-bind-text value="[[ $fields.departmentManager.$value()?.items[0]?.firstName ]]"></oj-bind-text></span>
      <span><oj-bind-text value="[[ $fields.departmentManager.$value()?.items[0]?.lastName ]]"></oj-bind-text></span>
      <br>
      <span><oj-bind-text value="[[ $fields.departmentManager.$value()?.items[0]?.phoneNumber ]]"></oj-bind-text></span>
    </div>
  </oj-list-item-layout>
</template>

Customize a Display Value for a Dropdown List

  1. Create an item layout managerItemLayout in the top-level layouts section of layout.json file 
      "managerItemLayout": {
    "description": "template layout for manager lov",
    "layoutType": "item",
      "layout": {
        "displayValue": [
          "firstName",
          "email""
        ]
      }
  },
  2. In the same layout.json file, under the display properties of the layout for the dynamic form, set the itemLayout property of the dropdown list field to the item layout managerItemLayout created in the previous step.
  3. Create a field template for manager field, and configure the Select Single component (oj-select-single).
    4. <template id="managerTemplate">
 <oj-select-single label-hint="[[ $labelHint ]]" value="{{ $value }}" data="[[ $variables.managerSDP ]]"
    item-text="[[$functions.createItemTextFunction($itemLayout.displayValue)]]"></oj-select-single>
</template>
  4. Create a function createItemTextFunction in the JavaScript tab of the dynamic layout to set the item-text property of the select-single component. 
        createItemTextFunction(fieldNames){
      return function(itemContext) {
        let ret = "";
          for (let i = 0; i < fieldNames.length; i++) {
            if (typeof fieldNames[i] === 'string') {
              if (i !== 0) {
                ret += ' : ';
              }
              ret += `${itemContext.data[fieldNames[i]]}`;
            } else if (typeof fieldNames[i] === 'object') {
              let ct = 0;
              for (let key in fieldNames[i]) {
                if (i !== 0 || ct !== 0) {
                  if (fieldNames[i][key].hasOwnProperty('separator')) {
                    ret += fieldNames[i][key].separator;
                  } else {
                    ret += ' : ';
                  }
                }
                ret += `${itemContext.data[key]}`;
                ct++;
              }
            }
          }
        return ret;
      };
    }
